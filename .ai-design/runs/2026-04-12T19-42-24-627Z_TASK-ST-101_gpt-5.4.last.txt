+void UGameTimeSubsystem::RunWeeklySimulationPhase(EMusicWeeklySimulationPhase Phase, const FDateTime& PreviousDate, const FDateTime& NewDate, bool bClosedMonth)
+{
+    LastExecutedWeeklyPhases.Add(Phase);
+
+    UE_LOG(LogMusicSimTime, Verbose, TEXT("Executing weekly simulation phase: %s"), ToPhaseName(Phase));
+
+    UGameInstance* GameInstance = GetGameInstance();
+    if (!GameInstance)
+    {
+        return;
+    }
+
+    switch (Phase)
+    {
+    case EMusicWeeklySimulationPhase::TrendDrift:
+        // Reserved for native weekly trend drift once the market trend model exists.
+        break;
+
+    case EMusicWeeklySimulationPhase::ArtistCondition:
+        if (bClosedMonth)
+        {
+            if (UArtistManagerSubsystem* ArtistSubsystem = GameInstance->GetSubsystem<UArtistManagerSubsystem>())
+            {
+                ArtistSubsystem->HandleMonthAdvanced(NewDate);
+            }
+        }
+        break;
+
+    case EMusicWeeklySimulationPhase::ProductionProgress:
+        // Current production logic is still month-end wrapped in the record subsystem.
+        break;
+
+    case EMusicWeeklySimulationPhase::ReleaseLaunchProcessing:
+        // Reserved for native weekly release launches.
+        break;
+
+    case EMusicWeeklySimulationPhase::MarketExposure:
+        if (bClosedMonth)
+        {
+            if (UMarketManagerSubsystem* MarketSubsystem = GameInstance->GetSubsystem<UMarketManagerSubsystem>())
+            {
+                MarketSubsystem->HandleMonthAdvanced(NewDate);
+            }
+        }
+        break;
+
+    case EMusicWeeklySimulationPhase::ChartCalculation:
+        if (bClosedMonth)
+        {
+            if (URecordManagerSubsystem* RecordSubsystem = GameInstance->GetSubsystem<URecordManagerSubsystem>())
+            {
+                RecordSubsystem->HandleMonthAdvanced(NewDate);
+            }
+        }
+        break;
+
+    case EMusicWeeklySimulationPhase::TourShowResolution:
+        // Reserved for touring once that subsystem lands.
+        break;
+
+    case EMusicWeeklySimulationPhase::FinancePosting:
+        if (bClosedMonth)
+        {
+            if (UFinanceManagerSubsystem* FinanceSubsystem = GameInstance->GetSubsystem<UFinanceManagerSubsystem>())
+            {
+                FinanceSubsystem->HandleMonthAdvanced(NewDate);
+            }
+        }
+        break;
+
+    case EMusicWeeklySimulationPhase::CriticNewsGeneration:
+    case EMusicWeeklySimulationPhase::UINotifications:
+        // Delegates fire after deterministic state mutation so listeners see settled state.
